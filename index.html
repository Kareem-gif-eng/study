المطلوب: إنشاء ملف باسم task.php داخل مجلد المشروع، وتعريف ثابث PASSING_GRADE بقيمة 50، وطباعة ناتج العمليات الحسابية لمتقاربين، ثم استخدام المعاملات المنطقية والمقارنة لطباعة النتيجة.

task.php 
<?php

define("PASSING_GRADE", 50);

// 2. إنشاء المتغيرين
$num1 = 12;
$num2 = 4;

// عرض نتائج العمليات الحسابية الأربع
echo "Sum: " . ($num1 + $num2) . "\n";
echo "Difference: " . ($num1 - $num2) . "\n";

$age = 20;

شرح وتحليل الكود:

• define() تُستخدم لإنشاء ثوابت لا يمكن تغيير قيمتها أثناء تنفيذ السكربت.

• الشرط الثلاثي (condition) ? true : false أسلوب مختصر لكتابة جملة if البسيطة.

• المعامل المنطقي && يضمن تحقق الشرطين معاً (أن تكون الدرجة أصلية أو تساوي 50 والعمر أكبر من 18).

Workshop 2 - Task 1

لعبة FizzBuzz مع التحقق من الأعداد الأولية (Prime Numbers)

المفاهيم: loops, conditionals, modulus %, string concatenation, prime detection function.

task1.php 
<?php
// دالة للتحقق مما إذا كان الرقم أولياً
function isPrime($n) {
    if ($n < 2) return false;
    for ($i = 2; $i <= sqrt($n); $i++) {
        if ($n % $i == 0) return false;
    }
    return true;
}

// التكرار من 1 إلى 50
for ($i = 1; $i <= 50; $i++) {
    if ($i % 3 == 0 && $i % 5 == 0) {
        $output = "FizzBuzz";
    } elseif ($i % 3 == 0) {
        $output = "Fizz";
    } elseif ($i % 5 == 0) {
        $output = "Buzz";
    } else {
        $output = $i;
    }

    // إذا كان الرقم أولياً نضع النتيجة بين أقواس مربعة []
    echo isPrime($i) ? "[$output]" : $output;
    echo "\n";
}
?>

تقرير درجات الطلاب مع فحص المدخلات (Grade Report with Validation)

المفاهيم: switch(true) pattern, is_numeric(), continue, type casting, input validation.

task2.php 
<?php
$scores = [95, "abc", 72, -5, 105, 60, "88"];

foreach ($scores as $score) {
    // 1. التحقق مما إذا كانت المدخلات عدداً
    if (!is_numeric($score)) {
        echo "Invalid entry: $score\n";
        continue;
    }

    $score = (float)$score;

    // 2. التحقق من المدى المقبول (من 0 إلى 100)
    if ($score < 0 || $score > 100) {
        echo "Out of range: $score\n";
        continue;
    }

    // 3. تصنيف الدرجات باستخدام خدعة switch(true)
    switch (true) {
        case $score >= 90:
            $grade = "A";
            break;
        case $score >= 75:
            $grade = "B";
            break;
        case $score >= 60:
            $grade = "C";
            break;
        default:
            $grade = "F";
    }

    echo "Score $score - $grade\n";
}
?>

حاسبة نصية مع معالجة الأخطاء (Simple Text Calculator)

المفاهيم: switch statement, Edge cases (Division by Zero), Default case for invalid operators.

task5.php 
<?php
$operations = [
    ["num1" => 10, "num2" => 5, "op" => "+"],
    ["num1" => 10, "num2" => 0, "op" => "/"],
    ["num1" => 7,  "num2" => 3, "op" => "*"],
    ["num1" => 9,  "num2" => 2, "op" => "%"],
    ["num1" => 5,  "num2" => 5, "op" => "^"] // معامل غير معروف
];

foreach ($operations as $operation) {
    $a  = $operation["num1"];
    $b  = $operation["num2"];
    $op = $operation["op"];

    switch ($op) {
        case "+":
            echo "$a + $b = " . ($a + $b) . "\n";
            break;
        case "-":
            echo "$a - $b = " . ($a - $b) . "\n";
            break;
        case "*":
            echo "$a * $b = " . ($a * $b) . "\n";
            break;
        case "/":
            if ($b == 0) {
                echo "Error: division by zero\n";
            } else {
                echo "$a / $b = " . ($a / $b) . "\n";
            }
            break;
        case "%":
            if ($b == 0) {
                echo "Error: division by zero\n";
            } else {
                echo "$a % $b = " . ($a % $b) . "\n";
            }
            break;
        default:
            echo "Error: unknown operator $op\n";

<?php
function formatUsername($fullName, $maxLength = 10) {
    // 1. تحويل الاسم إلى حروف صغيرة
    $formatted = strtolower($fullName);
    
    // 2. استبدال المسافات بالشرطة السفلية
    $formatted = str_replace(" ", "_", $formatted);
    
    // 3. اقتطاع النص إذا تجاوز الحد الأقصى
    if (strlen($formatted) > $maxLength) {
        $formatted = substr($formatted, 0, $maxLength) . "...";
    }
    
    return $formatted;
}

// اختبار الدالة

function addContact($book, $name, $phone) {

Workshop 4 - Task 3

التحقق الكامل من بيانات نموذج التسجيل (Server-Side Validation)

المفاهيم: empty(), strlen(), filter_var(), FILTER_VALIDATE_EMAIL, FILTER_VALIDATE_URL, Error Collection Pattern.

register.php 
<?php
$errors = [];
$success = false;

if ($_SERVER['REQUEST_METHOD'] === 'POST') {
    $name     = $_POST['name']     ?? '';
    $email    = $_POST['email']    ?? '';
    $age      = $_POST['age']      ?? '';
    $website  = $_POST['website']  ?? '';
    $password = $_POST['password'] ?? '';

    // 1. التحقق من الاسم
    if (empty($name) || strlen($name) < 2) {
        $errors[] = "Name must be at least 2 characters";
    }

    // 2. التحقق من البريد الإلكتروني
    if (empty($email) || !filter_var($email, FILTER_VALIDATE_EMAIL)) {
        $errors[] = "Please enter a valid email";
    }

    // 3. التحقق من العمر
    if (empty($age) || !is_numeric($age) || $age < 13 || $age > 120) {
        $errors[] = "Age must be between 13 and 120";
    }

    // 4. التحقق من الموقع (اختياري)
    if (!empty($website) && !filter_var($website, FILTER_VALIDATE_URL)) {
        $errors[] = "Website must be a valid URL";
    }

    // 5. التحقق من كلمة السر
    if (empty($password) || strlen($password) < 8) {
        $errors[] = "Password must be at least 8 characters";
    }

    if (empty($errors)) {
        $success = true;
    }
}
?>

إعداد قاعدة البيانات والجدول (MySQL Script)

المفاهيم: CREATE DATABASE, CREATE TABLE, Auto Increment, Default Timestamp.

db_setup.sql 
CREATE DATABASE workshop_db;
USE workshop_db;

CREATE TABLE products (
    id INT AUTO_INCREMENT PRIMARY KEY,
    name VARCHAR(100) NOT NULL,
    price DECIMAL(10,2) NOT NULL,
    quantity INT NOT NULL DEFAULT 0,
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

Workshop 5 - Task 2

عرض المنتجات وتطبيق الفلترة الديناميكية (Catalog Reader)

المفاهيم: Dynamic Queries, SELECT, mysqli_fetch_assoc(), WHERE, ORDER BY.

catalog.php 
<?php
require_once("connect.php");

$sql = "SELECT * FROM products WHERE 1=1";

if (isset($_GET['min_price']) && is_numeric($_GET['min_price'])) {
    $minPrice = $_GET['min_price'];
    $sql .= " AND price >= $minPrice";
}

if (isset($_GET['sort']) && $_GET['sort'] === 'price_desc') {
    $sql .= " ORDER BY price DESC";
} else {
    $sql .= " ORDER BY name ASC";
}

$result = mysqli_query($conn, $sql);

if (!$result) {
    die("Query failed: " . mysqli_error($conn));
}

$totalCount = 0;
$totalValue = 0;

while ($row = mysqli_fetch_assoc($result)) {
    echo $row['name'] . " - $" . $row['price'] . " (Qty: " . $row['quantity'] . ")\n";
    $totalCount++;
    $totalValue += $row['price'] * $row['quantity'];
}

echo "\nTotal products shown: $totalCount\n";
echo "Total inventory value shown: $" . number_format($totalValue, 2) . "\n";
?>
Workshop 5 - Task 3

تحديث مخزون المنتجات مع معالجة الشحنات الواردة

المفاهيم: UPDATE Statement, SELECT Check, mysqli_affected_rows().

restock.php 
<?php
require_once("connect.php");

$shipment = [
    "Laptop" => 5,
    "Mouse"  => 50,
    "Drone"  => 20 // عنصر غير موجود بالجدول لتقييم معالجة الاستثناء
];

foreach ($shipment as $name => $amount) {
    // 1. التحقق من وجود المنتج
    $checkSql = "SELECT * FROM products WHERE name = '$name'";
    $checkResult = mysqli_query($conn, $checkSql);

    if (mysqli_num_rows($checkResult) === 0) {
        echo "Skipped: $name (product not found)\n";
        continue;
    }

    // 2. تحديث الكمية بإضافتها للكمية السابقة
    $updateSql = "UPDATE products SET quantity = quantity + $amount WHERE name = '$name'";
    mysqli_query($conn, $updateSql);
    
    $affected = mysqli_affected_rows($conn);
    echo "Updated: $name, rows affected: $affected\n";
}

echo "\n--- Updated Table ---\n";
$result = mysqli_query($conn, "SELECT * FROM products");
while ($row = mysqli_fetch_assoc($result)) {
    echo $row['name'] . ": " . $row['quantity'] . " in stock\n";
}
?>
Workshop 5 - Task 4

حذف المنتجات ومناقشة تقنية الحذف الخفي (Soft Delete)

المفاهيم: DELETE Statement, Hard Delete vs Soft Delete, Data Preservation, Soft-Delete Strategy.

cleanup.php 
<?php
/*
 * Discussion:
 * Hard-deleting rows immediately is risky in production: data is gone permanently, 
 * which is dangerous if the delete was a mistake or if data is needed later for auditing.
 * A "soft delete" adds a column like is_deleted (boolean) or deleted_at (timestamp).
 * Instead of DELETE, you UPDATE that column, preserving data history while filtering it out from user views.
 */

require_once("connect.php");

// تصفير كمية أحد المنتجات لأغراض التجربة
mysqli_query($conn, "UPDATE products SET quantity = 0 WHERE name = 'Headphones'");

echo "--- Out of stock before delete ---\n";
$result = mysqli_query($conn, "SELECT * FROM products WHERE quantity = 0");
while ($row = mysqli_fetch_assoc($result)) {
    echo $row['name'] . "\n";
}

// تنفيذ الحذف للكميات الصفرية
mysqli_query($conn, "DELETE FROM products WHERE quantity = 0");
echo "\nRows deleted: " . mysqli_affected_rows($conn) . "\n";

$countResult = mysqli_query($conn, "SELECT COUNT(*) AS total FROM products");
$countRow = mysqli_fetch_assoc($countResult);
echo "Products remaining: " . $countRow['total'] . "\n";
?>
